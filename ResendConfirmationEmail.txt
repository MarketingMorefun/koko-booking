query ResendConfirmationEmail verb=POST {
  api_group = "KOKO Booking"

  input {
    text key
    text request_id?
    object payload {
      schema {
        int booking_id
      }
    }
  }

  stack {
    precondition ($env.ADMIN_KEY != null && $env.ADMIN_KEY != "" && $input.key == $env.ADMIN_KEY) {
      error_type = "accessdenied"
      error = "Unauthorized"
    }
  
    api.lambda {
      code = "return $input.request_id || crypto.randomBytes(16).toString('hex');"
      timeout = 5
    } as $RequestId
  
    function.run send_booking_confirmation {
      input = {
        booking_id: $input.payload.booking_id
        automatic : false
        request_id: $RequestId
      }
    } as $Result
  }

  response = $Result
}
