                        |push:"Authorization: Bearer re_YOUR_RESEND_API_KEY"
                        |push:"Content-Type: application/json"
                    } as $ResendEmail
                  
                    db.patch bookings {
                      field_name = "id"
                      field_value = $Booking.id
                      data = {confirmation_email_sent_at: $now_ts}
                    } as $Booking
                  
                    var $message {
                      value = "Booking marked as deposit paid and confirmation email sent"
                    }
                  }
                
                  else {
                    var $message {
                      value = "Booking marked as deposit paid, confirmation email already sent"
                    }
                  }
                }
              }
            
              elseif ($event_type == "checkout.session.expired") {
                conditional {
                  if ($Booking.status != "deposit_paid") {
                    db.patch bookings {
                      field_name = "id"
                      field_value = $Booking.id
                      data = {status: "expired"}
                    } as $Booking
                  
                    var $message {
                      value = "Booking marked as expired"
                    }
                  }
                
                  else {
                    var $message {
                      value = "Booking already deposit paid, expired event ignored"
                    }
                  }
                }
              }
            
              else {
                var $message {
                  value = "Event ignored"
                }
              }
            }
          }
        }
      }
    }
  }

  response = {
    received     : true
    error_message: $error_message
    message      : $message
    event_type   : $event_type
    booking_id   : $booking_id
    booking      : $Booking
    package      : $Package
    resend_email : $ResendEmail
  }
}
