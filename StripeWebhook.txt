// Authenticate the exact bytes before trusting any event data or returning PII.
query StripeWebhook verb=POST {
  api_group = "KOKO Booking"

  input {
  }

  stack {
    util.get_raw_input {
      encoding = "none"
      exclude_middleware = true
    } as $raw_body
  
    db.query stripe_location_keys {
      return = {type: "list"}
    } as $StripeKeys
  
    api.lambda {
      code = """
        const raw = $var.raw_body;
        const headers = $env.$http_headers || {};
        const header = Object.entries(headers).find(([k]) => k.toLowerCase() === 'stripe-signature')?.[1];
        if (typeof raw !== 'string' || typeof header !== 'string') return {ok:false};
        const fields = header.split(',').map(s => s.trim().split('='));
        const timestamp = fields.find(([k]) => k === 't')?.[1];
        if (!/^\d+$/.test(timestamp || '') || Math.abs(Date.now()/1000-Number(timestamp)) > 300) return {ok:false};
        const candidates = fields.filter(([k,v]) => k === 'v1' && /^[a-f0-9]{64}$/.test(v || '')).map(([,v]) => v);
        const locationIds = [];
        for (const key of $var.StripeKeys || []) {
          if (!key.is_active || !key.webhook_secret_live) continue;
          const expected = crypto.createHmac('sha256', key.webhook_secret_live).update(timestamp+'.'+raw).digest('hex');
          if (candidates.some(s => crypto.timingSafeEqual(Buffer.from(s,'hex'),Buffer.from(expected,'hex')))) locationIds.push(key.location_id);
        }
        if (!locationIds.length) return {ok:false};
        try { return {ok:true, event:JSON.parse(raw), location_ids:locationIds}; } catch { return {ok:false}; }
        """
      timeout = 10
    } as $Verified
  
    precondition ($Verified.ok) {
      error_type = "accessdenied"
      error = "Invalid Stripe webhook signature"
    }
  
    var $event {
      value = $Verified.event
    }
  
    var $session {
      value = $event|get:"data"|get:"object"
    }
  
    conditional {
      if ($event.type != "checkout.session.completed" && $event.type != "checkout.session.async_payment_succeeded" && $event.type != "checkout.session.expired") {
        return {
          value = {received: true, message: "Event ignored"}
        }
      }
    }
  
    // Shared Stripe accounts also send Cityheroes events. Only this booking
    // integration's signed checkout metadata may select a KOKO booking.
    conditional {
      if ($session.metadata.stripe_mode != "live" || ($session.metadata.payment_type != "deposit" && $session.metadata.payment_type != "full")) {
        return {
          value = {
            received: true
            message : "Unrelated integration ignored"
          }
        }
      }
    }
  
    db.get bookings {
      field_name = "id"
      field_value = $session
        |get:"metadata"
        |get:"booking_id"
        |to_int
    } as $Booking
  
    conditional {
      if ($Booking == null) {
        return {
          value = {received: true, message: "Unrelated checkout ignored"}
        }
      }
    }
  
    api.lambda {
      code = "return $var.Verified.location_ids.includes($var.Booking.location_id) && $var.event.livemode === true;"
      timeout = 10
    } as $CorrectAccount
  
    precondition ($CorrectAccount) {
      error_type = "accessdenied"
      error = "Stripe account does not match booking"
    }
  
    db.get stripe_location_keys {
      field_name = "location_id"
      field_value = $Booking.location_id
    } as $StripeKey
  
    // Read Stripe's current state; signed events can arrive out of order.
    api.request {
      url = "https://api.stripe.com/v1/checkout/sessions/"|concat:$session.id:""
      method = "GET"
      headers = []
        |push:("Authorization: Bearer "
          |concat:$StripeKey.stripe_secret_key_live:""
        )
    } as $StripeResponse
  
    precondition ($StripeResponse.response.status == 200) {
      error = "Stripe session could not be verified; retry delivery"
    }
  
    function.run apply_stripe_checkout {
      input = {
        booking_id: $Booking.id
        session   : $StripeResponse.response.result
        event_id  : $event.id
        event_type: $event.type
      }
    } as $Result
  
    conditional {
      if ($Result.action == "settle" || $Result.action == "duplicate") {
        function.run fulfil_paid_booking {
          input = {booking_id: $Booking.id}
        } as $Fulfilment
      }
    }
  }

  response = {received: true}
}
