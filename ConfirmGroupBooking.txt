query ConfirmGroupBooking verb=POST {
  api_group = "KOKO Booking"

  input {
    object payload {
      schema {
        int booking_id
        text payment_option?
      }
    }
  }

  stack {
    // Callers that only send booking_id (the /booking/resume page linked from
    // reminder emails) must still get a deposit checkout. Passing the optional
    // field straight through left the function input empty ("Missing param").
    var $payment_option {
      value = "deposit"
    }
  
    conditional {
      if ($input.payload.payment_option == "full") {
        var $payment_option {
          value = "full"
        }
      }
    }
  
    function.run create_booking_checkout {
      input = {
        booking_id    : $input.payload.booking_id
        payment_option: $payment_option
        booking_type  : "group_booking"
      }
    } as $Result
  }

  response = $Result
}
