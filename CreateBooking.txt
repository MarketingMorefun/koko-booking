query CreateBooking verb=POST {
  api_group = "KOKO Booking"

  input {
    object payload {
      schema {
        text location_slug filters=trim
        int party_room_id
        date date
        int guests
        int package_id
        object[] addons? {
          schema {
            int addon_id?
            int qty?
            text option?
            text option_label?
            text allergy_note?
          }
        }

        text customer_name?
        text customer_phone?
        text customer_email?
        text birthday_child_name?
        text birthday_child_gender?
        text average_age?
        text booking_notes?
        text referral_code?
        timestamp start_ts
        timestamp end_ts
      }
    }
  }

  stack {
    // 顺手清理过期 hold,保持订单表干净(Free Plan 没定时任务的替代做法)
    function.run expire_stale_holds as $expired_result

    var $error_message {
      value = null
    }

    var $Booking {
      value = null
    }

    var $Locations1 {
      value = null
    }

    var $Room {
      value = null
    }

    var $package_total_cents {
      value = 0
    }

    var $addons_total_cents {
      value = 0
    }

    var $grand_total_cents {
      value = 0
    }

    var $addons_json {
      value = []
    }

    var $now_ts {
      value = `"now"|to_timestamp:"Australia/Sydney"`
    }

    var $min_bookable_ts {
      value = `$now_ts|add_ms_to_timestamp:259200000`
    }

    var $hold_expires_at_ts {
      value = `$now_ts|add_ms_to_timestamp:900000`
    }

    var $guests_int {
      value = `$input.payload.guests|first_notnull:0|to_int`
    }

    var $req_addons {
      value = $input.payload.addons|first_notnull:[]
    }

    var $start_date_from_ts {
      value = `$input.payload.start_ts|format_timestamp:"Y-m-d":"Australia/Sydney"`
    }

    var $end_date_from_ts {
      value = `$input.payload.end_ts|format_timestamp:"Y-m-d":"Australia/Sydney"`
    }

    // ---- Referral / repeat-customer $30 discount ----
    // Vars only — the actual lookup + reservation happens later, right
    // before db.add, once every validation check has passed. Reserving a
    // credit before we're sure a booking will actually be created would
    // strand it forever with no booking row around to ever release it.
    var $customer_email_normalised {
      value = $input.payload.customer_email|trim|lower
    }

    var $discount_cents {
      value = 0
    }

    var $discount_reason {
      value = ""
    }

    var $applied_credit_id {
      value = 0
    }

    var $referral_code_used {
      value = ""
    }

    var $AvailableCredit {
      value = null
    }

    db.get Locations {
      field_name = "slug"
      field_value = $input.payload.location_slug
    } as $Locations1

    conditional {
      if ($Locations1 == null) {
        var $error_message {
          value = "Location not found"
        }
      }

      elseif ($guests_int < 10) {
        var $error_message {
          value = "Party bookings require a minimum of 10 guests"
        }
      }

      elseif ($input.payload.start_ts == null || $input.payload.end_ts == null) {
        var $error_message {
          value = "Missing booking time"
        }
      }

      elseif ($start_date_from_ts != $input.payload.date) {
        var $error_message {
          value = "Booking date does not match selected start time"
        }
      }

      elseif ($end_date_from_ts != $input.payload.date) {
        var $error_message {
          value = "Booking date does not match selected end time"
        }
      }

      elseif ($input.payload.start_ts < $min_bookable_ts) {
        var $error_message {
          value = "Bookings must be made at least 72 hours in advance"
        }
      }

      elseif ($input.payload.end_ts <= $input.payload.start_ts) {
        var $error_message {
          value = "Invalid booking time"
        }
      }
    }

    conditional {
      if ($error_message == null) {
        db.get party_rooms {
          field_name = "id"
          field_value = $input.payload.party_room_id
        } as $Room

        conditional {
          if ($Room == null) {
            var $error_message {
              value = "Party room not found"
            }
          }

          elseif ($Room.Location_id != $Locations1.id) {
            var $error_message {
              value = "Party room does not belong to this location"
            }
          }

          elseif ($Room.is_active != true) {
            var $error_message {
              value = "Party room is inactive"
            }
          }

          elseif ($guests_int < $Room.capacity_min || $guests_int > $Room.capacity_max) {
            var $error_message {
              value = "Guest count is outside this room capacity"
            }
          }
        }
      }
    }

    conditional {
      if ($error_message == null) {
        db.query bookings {
          where = $db.bookings.location_id == $Locations1.id && $db.bookings.party_room_id == $input.payload.party_room_id && $db.bookings.start_ts < $input.payload.end_ts && $db.bookings.end_ts > $input.payload.start_ts && ($db.bookings.status == "pending_payment" || $db.bookings.status == "deposit_paid" || $db.bookings.status == "paid" || $db.bookings.status == "confirmed" || ($db.bookings.status == "hold" && $db.bookings.hold_expires_at_ts > $now_ts))
          return = {type: "list"}
        } as $slot_conflicts

        conditional {
          if ($slot_conflicts[0] != null) {
            var $error_message {
              value = "Selected slot is no longer available"
            }
          }
        }
      }
    }

    conditional {
      if ($error_message == null) {
        db.query package_location_price {
          where = $db.package_location_price.location_id == $Locations1.id && $db.package_location_price.package_id == $input.payload.package_id && $db.package_location_price.is_active == true
          return = {type: "list"}
        } as $plp_list

        array.find ($plp_list) if ($this.id > 0) as $plp
        db.query package_location_rules {
          where = $db.package_location_rules.location_id == $Locations1.id && $db.package_location_rules.package_id == $input.payload.package_id && $db.package_location_rules.is_active == true
          return = {type: "list"}
        } as $rule_list

        array.find ($rule_list) if ($this.id > 0) as $rule
        conditional {
          if ($plp == null) {
            var $error_message {
              value = "Package is not available for this location"
            }
          }

          elseif ($rule == null) {
            var $error_message {
              value = "Package rules not found for this location"
            }
          }

          elseif ($guests_int < ($rule.min_guests|first_notnull:0|to_int)) {
            var $error_message {
              value = "Guest count does not meet package minimum"
            }
          }

          elseif (($rule.max_guests|first_notnull:null) != null && $guests_int > ($rule.max_guests|to_int)) {
            var $error_message {
              value = "Guest count exceeds package maximum"
            }
          }

          else {
            var $price_pp_cents_calc {
              value = `$plp.price_pp_cents|first_notnull:0|to_int`
            }

            var $package_total_cents {
              value = `$guests_int|multiply:$price_pp_cents_calc`
            }
          }
        }
      }
    }

    conditional {
      if ($error_message == null) {
        db.query location_addons {
          where = $db.location_addons.location_id == $Locations1.id && $db.location_addons.is_active == true
          return = {type: "list"}
        } as $loc_addon_list

        array.map ($req_addons) {
          by = {
            addon_id    : `$this.addon_id|first_notnull:0|to_int`
            qty         : `$this.qty|first_notnull:1|to_int`
            option      : $this.option
            option_label: $this.option_label
            allergy_note: $this.allergy_note
          }
        } as $addon_lines

        array.map ($addon_lines) {
          by = $this.addon_id
        } as $addon_ids_int

        db.query Addons {
          where = $db.Addons.id in $addon_ids_int && $db.Addons.is_active == true
          return = {type: "list"}
        } as $addon_defs

        foreach ($addon_lines) {
          each as $item {
            array.filter ($loc_addon_list) if ($this.addon_id == $item.addon_id) as $matched_loc_addon_list
            array.find ($matched_loc_addon_list) if ($this.id > 0) as $matched_loc_addon
            array.filter ($addon_defs) if ($this.id == $item.addon_id) as $matched_addon_def_list
            array.find ($matched_addon_def_list) if ($this.id > 0) as $matched_addon_def
            conditional {
              if ($matched_loc_addon == null || $matched_addon_def == null) {
                var $error_message {
                  value = "Invalid addon for this location"
                }
              }
            }

            conditional {
              if ($error_message == null) {
                var $unit_type {
                  value = `$matched_addon_def.unit_type|first_notnull:"per_item"`
                }

                var $qty_num {
                  value = `$item.qty|first_notnull:1|to_int`
                }

                conditional {
                  if ($qty_num < 1) {
                    var $qty_num {
                      value = 1
                    }
                  }
                }

                var $price_num {
                  value = `$matched_loc_addon.default_price_cents|first_notnull:0|to_int`
                }

                conditional {
                  if ($price_num == 0 && $matched_loc_addon.default_price != null) {
                    var $price_num {
                      value = ($matched_loc_addon.default_price|multiply:100)|round:0
                    }
                  }
                }

                conditional {
                  if ($unit_type == "per_person") {
                    var $line_total_cents {
                      value = $price_num|multiply:$guests_int
                    }
                  }

                  else {
                    var $line_total_cents {
                      value = $price_num|multiply:$qty_num
                    }
                  }
                }

                var $addons_total_cents {
                  value = `$addons_total_cents|add:$line_total_cents`
                }

                array.push $addons_json {
                  value = {
                    addon_id        : $matched_addon_def.id
                    addon_name      : $matched_addon_def.name
                    qty             : $qty_num
                    unit_type       : $unit_type
                    unit_price_cents: $price_num
                    line_total_cents: $line_total_cents
                    option          : $item.option
                    option_label    : $item.option_label
                    allergy_note    : $item.allergy_note
                  }
                }
              }
            }
          }
        }
      }
    }

    conditional {
      if ($error_message == null) {
        var $grand_total_cents {
          value = ($package_total_cents|first_notnull:0|to_int) + ($addons_total_cents|first_notnull:0|to_int)
        }

        // ---- Referral / repeat-customer $30 discount ----
        // Placed here (all validation passed, right before db.add) rather
        // than at the top of the stack — once a credit is reserved
        // (flipped to "pending"), a booking row MUST get created so
        // expire_stale_holds has something to release it from later if
        // it's abandoned. Reserving to "pending" (not leaving it
        // "available") also means a second concurrent hold from the same
        // customer can't find and double-spend the same credit.
        db.query customer_credits {
          where = $db.customer_credits.customer_email == $customer_email_normalised && $db.customer_credits.status == "available"
          return = {type: "single"}
        } as $AvailableCredit

        conditional {
          if ($AvailableCredit != null) {
            var $discount_cents {
              value = 3000
            }

            var $discount_reason {
              value = "repeat_customer"
            }

            var $applied_credit_id {
              value = $AvailableCredit.id
            }

            db.patch customer_credits {
              field_name = "id"
              field_value = $AvailableCredit.id
              data = {status: "pending"}
            } as $ReservedCredit
          }

          elseif ($input.payload.referral_code != null && $input.payload.referral_code != "") {
            db.query bookings {
              where = $db.bookings.referral_code == $input.payload.referral_code && ($db.bookings.status == "deposit_paid" || $db.bookings.status == "paid")
              return = {type: "single"}
            } as $ReferringBooking

            conditional {
              if ($ReferringBooking != null && ($ReferringBooking.customer_email|trim|lower) != $customer_email_normalised) {
                var $discount_cents {
                  value = 3000
                }

                var $discount_reason {
                  value = "referral"
                }

                var $referral_code_used {
                  value = $input.payload.referral_code
                }
              }
            }
          }
        }

        var $balance_due_cents {
          value = $grand_total_cents - $discount_cents
        }

        conditional {
          if ($balance_due_cents < 0) {
            var $balance_due_cents {
              value = 0
            }
          }
        }

        db.add bookings {
          enforce_hidden_fields = false
          data = {
            location_id               : $Locations1.id
            party_room_id             : $input.payload.party_room_id
            date                      : $input.payload.date
            start_ts                  : $input.payload.start_ts
            end_ts                    : $input.payload.end_ts
            guests                    : $guests_int
            status                    : "hold"
            customer_name             : $input.payload.customer_name
            customer_phone            : $input.payload.customer_phone
            customer_email            : $input.payload.customer_email
            hold_expires_at_ts        : $hold_expires_at_ts
            addons_json               : $addons_json
            package_total_cents       : $package_total_cents
            addons_total_cents        : $addons_total_cents
            grand_total_cents         : $grand_total_cents
            package_id                : $input.payload.package_id
            stripe_session_id         : ""
            payment_url               : ""
            paid_at                   : null
            confirmation_email_sent_at: null
            birthday_child_name       : $input.payload.birthday_child_name
            birthday_child_gender     : $input.payload.birthday_child_gender
            average_age               : $input.payload.average_age
            booking_notes             : $input.payload.booking_notes
            deposit_paid_cents        : 0
            balance_due_cents         : $balance_due_cents
            discount_cents            : $discount_cents
            discount_reason           : $discount_reason
            applied_credit_id         : $applied_credit_id
            referral_code_used        : $referral_code_used
            referral_code             : ""
            booking_type              : "birthday_party"
          }
        } as $Booking
      }
    }
  }

  response = {
    error_message  : $error_message
    booking        : $Booking
    discount_cents : $discount_cents
    discount_reason: $discount_reason
  }
}
