function send_booking_confirmation {
  input {
    int booking_id
    bool automatic
    text request_id
  }

  stack {
    db.transaction {
      stack {
        var $error_message {
          value = null
        }
      
        var $message {
          value = null
        }
      
        var $Booking {
          value = null
        }
      
        var $Package {
          value = null
        }
      
        var $BookingLocation {
          value = null
        }
      
        var $ResendEmail {
          value = null
        }
      
        var $child_name {
          value = ""
        }
      
        var $email_heading {
          value = "Your KOKO booking is confirmed!"
        }
      
        var $email_subject {
          value = "KOKO Booking Deposit Confirmation"
        }
      
        var $email_html {
          value = ""
        }
      
        var $package_name {
          value = ""
        }
      
        var $package_total_display {
          value = ""
        }
      
        var $addons_total_display {
          value = ""
        }
      
        var $grand_total_display {
          value = ""
        }
      
        var $deposit_paid_display {
          value = "$50.00"
        }
      
        var $remaining_balance_cents {
          value = 0
        }
      
        var $remaining_balance_display {
          value = ""
        }
      
        var $booking_start_display {
          value = ""
        }
      
        var $booking_end_display {
          value = ""
        }
      
        var $booking_time_display {
          value = ""
        }
      
        var $location_display {
          value = "KOKO Amusement"
        }
      
        var $package_display {
          value = ""
        }
      
        var $min_guest_note {
          value = ""
        }
      
        var $referral_code {
          value = ""
        }
      
        var $referral_card_url {
          value = ""
        }
      
        var $pkg_cents {
          value = 0
        }
      
        var $add_cents {
          value = 0
        }
      
        var $grand_cents {
          value = 0
        }
      
        var $deposit_cents {
          value = 0
        }
      
        var $now_ts {
          value = `"now"|to_timestamp:"Australia/Sydney"`
        }
      
        db.get bookings {
          field_name = "id"
          field_value = $input.booking_id
          lock = true
        } as $Booking
      
        conditional {
          if ($Booking != null && $input.automatic && (($Booking.confirmation_email_sent_at != null && ($Booking.confirmation_email_status == "accepted" || $Booking.confirmation_email_status == "delivered" || $Booking.confirmation_email_status == null || $Booking.confirmation_email_status == "")) || $Booking.confirmation_email_status == "bounced" || $Booking.confirmation_email_status == "failed_delivery")) {
            return {
              value = {
                error_message: null
                message      : "No automatic resend required"
              }
            }
          }
        
          elseif ($Booking != null && $input.automatic && ($Booking.confirmation_email_attempts|first_notnull:0) >= 5 && $Booking.confirmation_email_status == "failed") {
            return {
              value = {
                error_message: "Automatic confirmation retries exhausted"
              }
            }
          }
        
          elseif ($Booking != null && $input.automatic && $Booking.confirmation_email_next_retry_at != null && $Booking.confirmation_email_next_retry_at > $now_ts) {
            return {
              value = {error_message: null, message: "Retry not due"}
            }
          }
        
          elseif ($Booking != null && $Booking.status != "deposit_paid" && $Booking.status != "paid") {
            return {
              value = {
                error_message: "Only paid bookings can receive a payment confirmation"
              }
            }
          }
        }
      
        conditional {
          if ($Booking == null) {
            var $error_message {
              value = "Booking not found"
            }
          
            var $message {
              value = "Booking not found"
            }
          }
        
          elseif ($Booking.customer_email == null || $Booking.customer_email == "") {
            db.patch bookings {
              field_name = "id"
              field_value = $Booking.id
              data = {
                confirmation_email_status         : "failed"
                confirmation_email_error          : "Missing customer email"
                confirmation_email_attempts       : ($Booking.confirmation_email_attempts|first_notnull:0) + 1
                confirmation_email_last_attempt_at: $now_ts
                confirmation_email_next_retry_at  : $now_ts|add_ms_to_timestamp:300000
              }
            } as $Booking
          
            var $error_message {
              value = "Missing customer email"
            }
          
            var $message {
              value = "Customer email is missing, cannot send"
            }
          }
        
          else {
            var $pkg_cents {
              value = $Booking.package_total_cents|first_notempty:0
            }
          
            var $add_cents {
              value = $Booking.addons_total_cents|first_notempty:0
            }
          
            var $grand_cents {
              value = $Booking.grand_total_cents|first_notempty:0
            }
          
            var $deposit_cents {
              value = $Booking.deposit_paid_cents|first_notempty:0
            }
          
            var $remaining_balance_cents {
              value = $grand_cents - $deposit_cents - ($Booking.discount_cents|first_notnull:0)
            }
          
            conditional {
              if ($remaining_balance_cents < 0) {
                var $remaining_balance_cents {
                  value = 0
                }
              }
            }
          
            var $referral_code {
              value = $Booking.referral_code|first_notnull:""
            }
          
            conditional {
              if ($referral_code != "") {
                // CHANGE THIS if the referral card page ends up at a
                // different path than /booking/referral-card.
                var $referral_card_url {
                  value = "https://booking.morefun.com.au/booking/referral-card?code="|concat:$referral_code:""
                }
              }
            }
          
            conditional {
              if ($Booking.package_id != null && $Booking.package_id != "") {
                db.get Packages {
                  field_name = "id"
                  field_value = $Booking.package_id
                } as $Package
              }
            }
          
            var $email_subject {
              value = "KOKO Booking Deposit Confirmation #"|concat:$Booking.id:""
            }
          
            conditional {
              if ($Booking.birthday_child_name != null && $Booking.birthday_child_name != "") {
                var $child_name {
                  value = $Booking.birthday_child_name
                }
              
                var $email_heading {
                  value = $child_name
                    |concat:"'s birthday party is ready!":""
                }
              
                var $email_subject {
                  value = $child_name
                    |concat:"'s KOKO Birthday Party Confirmed #":""
                }
              
                var $email_subject {
                  value = $email_subject|concat:$Booking.id:""
                }
              }
            }
          
            var $package_total_display {
              value = "$"
                |concat:(($pkg_cents / 100)|number_format:2:".":","):""
            }
          
            var $addons_total_display {
              value = "$"
                |concat:(($add_cents / 100)|number_format:2:".":","):""
            }
          
            var $grand_total_display {
              value = "$"
                |concat:(($grand_cents / 100)|number_format:2:".":","):""
            }
          
            var $deposit_paid_display {
              value = "$"
                |concat:(($deposit_cents / 100)|number_format:2:".":","):""
            }
          
            var $remaining_balance_display {
              value = "$"
                |concat:(($remaining_balance_cents / 100)|number_format:2:".":","):""
            }
          
            conditional {
              if ($Booking.start_ts != null && $Booking.start_ts != "") {
                var $booking_start_display {
                  value = $Booking.start_ts
                    |to_timestamp:"Australia/Sydney"
                    |format_timestamp:"D, M j, Y H:i":"Australia/Sydney"
                }
              }
            }
          
            conditional {
              if ($Booking.end_ts != null && $Booking.end_ts != "") {
                var $booking_end_display {
                  value = $Booking.end_ts
                    |to_timestamp:"Australia/Sydney"
                    |format_timestamp:"H:i":"Australia/Sydney"
                }
              }
            }
          
            conditional {
              if ($booking_start_display != "" && $booking_end_display != "") {
                var $booking_time_display {
                  value = $booking_start_display|concat:" - ":""
                }
              
                var $booking_time_display {
                  value = $booking_time_display|concat:$booking_end_display:""
                }
              }
            
              elseif ($booking_start_display != "") {
                var $booking_time_display {
                  value = $booking_start_display
                }
              }
            }
          
            conditional {
              if ($Booking.location_id != null && $Booking.location_id != "") {
                db.get Locations {
                  field_name = "id"
                  field_value = $Booking.location_id
                } as $BookingLocation
              
                conditional {
                  if ($BookingLocation != null) {
                    var $location_display {
                      value = $BookingLocation.Name
                    }
                  }
                }
              }
            }
          
            var $package_display {
              value = $Booking.package_id
            }
          
            conditional {
              if ($Package != null) {
                var $package_name {
                  value = $Package|get:"Name"
                }
              
                conditional {
                  if ($package_name == null || $package_name == "") {
                    var $package_name {
                      value = $Package|get:"name"
                    }
                  }
                }
              
                conditional {
                  if ($package_name != null && $package_name != "") {
                    var $package_display {
                      value = $package_name
                    }
                  }
                }
              }
            }
          
            conditional {
              if ($Booking.booking_type == "group_booking") {
                var $min_guest_note {
                  value = "<div class='kk-box' style='background:#FFF7EB;border:1px solid #F4D7B5;border-radius:18px;padding:22px;margin:24px 0;'><p style='margin:0 0 0px;color:#7b6a58;font-size:14px;line-height:1.6;'><strong style='color:#2F241C;'>Minimum charge:</strong> Group bookings are charged for a minimum of 10 guests. If fewer than 10 guests attend on the day, your booking will still be charged at the 10-guest rate.</p></div>"
                }
              }
            }
          
            // Payment Summary table shows Grand Total then straight to
            // Deposit Paid / Remaining Balance — with no line item for the
            // discount, a customer doing the maths in their head (Grand
            // Total − Deposit) won't match Remaining Balance and will think
            // something's wrong. Insert an explicit row.
            var $discount_row {
              value = ""
            }
          
            conditional {
              if (($Booking.discount_cents|first_notnull:0) > 0) {
                var $discount_label {
                  value = ($Booking.discount_reason == "referral") ? "Referral Discount" : "Welcome Back Discount"
                }
              
                var $discount_display {
                  value = "-$"
                    |concat:(($Booking.discount_cents / 100)|number_format:2:".":","):""
                }
              
                var $discount_row {
                  value = "<tr><td style='padding:9px 0;color:#2F8F5B;font-weight:800;'>"
                    |concat:$discount_label:""
                    |concat:"</td><td style='padding:9px 0;text-align:right;font-weight:900;color:#2F8F5B;'>":""
                    |concat:$discount_display:""
                    |concat:"</td></tr>":""
                }
              }
            }
          
            // Itemised add-ons for the email (name, chosen option, qty x unit, line total).
            var $addon_rows {
              value = ""
            }
          
            foreach ($Booking.addons_json|first_notnull:[]) {
              each as $a {
                var $a_name {
                  value = $a
                    |get:"addon_name":""
                    |first_notnull:""
                    |replace:"<":"&lt;"
                }
              
                var $a_label {
                  value = $a
                    |get:"option_label":""
                    |first_notnull:""
                    |replace:"<":"&lt;"
                }
              
                var $a_qty {
                  value = $a|get:"qty":1|first_notnull:1
                }
              
                var $a_unit {
                  value = $a
                    |get:"unit_price_cents":0
                    |first_notnull:0
                }
              
                var $a_line {
                  value = $a
                    |get:"line_total_cents":0
                    |first_notnull:0
                }
              
                var $a_unit_display {
                  value = "$"
                    |concat:(($a_unit / 100)|number_format:2:".":","):""
                }
              
                var $a_line_display {
                  value = "$"
                    |concat:(($a_line / 100)|number_format:2:".":","):""
                }
              
                var $a_label_html {
                  value = ""
                }
              
                conditional {
                  if ($a_label != "") {
                    var $a_label_html {
                      value = "<div style='font-size:13px;color:#7b6a58;font-weight:400;margin-top:3px;'>"
                        |concat:$a_label:""
                        |concat:"</div>":""
                    }
                  }
                }
              
                var $addon_rows {
                  value = $addon_rows
                    |concat:"<tr><td style='padding:10px 0;border-top:1px solid #E8DDCC;color:#2F241C;vertical-align:top;'><strong style='font-weight:800;'>":""
                    |concat:$a_name:""
                    |concat:"</strong>":""
                    |concat:$a_label_html:""
                    |concat:"<div style='font-size:13px;color:#9A8B7C;margin-top:3px;'>":""
                    |concat:$a_qty:""
                    |concat:" × ":""
                    |concat:$a_unit_display:""
                    |concat:"</div></td><td style='padding:10px 0;border-top:1px solid #E8DDCC;text-align:right;font-weight:800;color:#2F241C;vertical-align:top;white-space:nowrap;'>":""
                    |concat:$a_line_display:""
                    |concat:"</td></tr>":""
                }
              }
            }
          
            var $addons_block {
              value = ""
            }
          
            conditional {
              if ($addon_rows != "") {
                var $addons_block {
                  value = "<div class='kk-box' style='background:#faf6ee;border:1px solid #E8DDCC;border-radius:18px;padding:22px;margin:24px 0;'><h2 style='margin:0 0 6px;font-size:18px;color:#2F241C;'>Add-ons</h2><table role='presentation' style='width:100%;border-collapse:collapse;font-size:15px;'>"
                    |concat:$addon_rows:""
                    |concat:"</table></div>":""
                }
              }
            }
          
            // The printable birthday invitation only suits birthday parties.
            var $invite_block {
              value = "<div class='kk-box' style='background:#ffffff;border:1px solid #E8DDCC;border-radius:18px;padding:22px;margin:24px 0;'><div style='text-align:center;'><div style='margin-bottom:14px;'><div style='display:inline-block;padding:9px 16px;border-radius:999px;background:#FFF3DF;color:#B86816;font-size:12px;font-weight:800;letter-spacing:0.12em;text-transform:uppercase;line-height:1;'>Party Invitation</div></div><h2 style='margin:0 0 10px;font-size:20px;line-height:1.2;color:#2F241C;'>Invite your guests</h2><p style='margin:0 0 16px;color:#7b6a58;font-size:14px;line-height:1.6;'>Download the KOKO birthday invitation, print it, then fill in the location, date and time for your party.</p><img src='https://cdn.jsdelivr.net/gh/MarketingMorefun/koko-booking@ec8c576/koko-birthday-invitation.png' alt='KOKO birthday invitation' width='240' style='width:240px;max-width:100%;height:auto;border-radius:14px;border:1px solid #E8DDCC;display:block;margin:0 auto 16px;'><a href='https://cdn.jsdelivr.net/gh/MarketingMorefun/koko-booking@ec8c576/koko-birthday-invitation.png' style='display:inline-block;background:#F5D949;border:2px solid #221D16;box-shadow:3px 3px 0 #221D16;color:#221D16;text-decoration:none;font-size:14px;font-weight:900;padding:12px 18px;border-radius:12px;margin:4px;'>Download PNG</a><a href='https://cdn.jsdelivr.net/gh/MarketingMorefun/koko-booking@ec8c576/koko-birthday-invitation.pdf' style='display:inline-block;background:#FFF7EB;color:#B86816;text-decoration:none;font-size:14px;font-weight:900;padding:12px 18px;border-radius:12px;border:2px solid #F2B36F;margin:4px;'>Printable PDF</a><p style='margin:12px 0 0;color:#9A8B7C;font-size:12px;'>Tip: print at A4 for the best result.</p></div></div>"
            }
          
            conditional {
              if ($Booking.booking_type == "group_booking") {
                var $invite_block {
                  value = ""
                }
              }
            }
          
            var $email_html {
              value = "<!doctype html><html lang='en'><head><meta charset='utf-8'><meta name='viewport' content='width=device-width, initial-scale=1'><style>@media only screen and (max-width: 520px) { .kk-txt { text-align: center !important; } .kk-outer { padding: 16px 8px !important; } .kk-card { padding: 20px 14px 26px !important; } .kk-box { padding: 18px 14px !important; } }</style></head><body style='margin:0;padding:0;background:#FFFDF9;'><div class='kk-outer' style='max-width:680px;margin:0 auto;padding:28px 20px;background:#FFFDF9;font-family:Arial,sans-serif;color:#2F241C;'><div class='kk-card' style='background:#ffffff;border:1px solid #E8DDCC;border-radius:24px;padding:28px 24px 34px;'><div class='kk-hero' style='background:#FFF4CC;border-radius:20px;padding:22px 16px 0;margin:0 0 26px;text-align:center;overflow:hidden;'><div style='display:inline-block;background:#ffffff;border:2px solid #221D16;border-radius:16px;padding:9px 18px;font-weight:900;font-size:16px;color:#221D16;box-shadow:3px 3px 0 #221D16;'>Woohoo! The remaining balance is payable on the day of your event.</p></div><div class='kk-box' style='background:#faf6ee;border:1px solid #E8DDCC;border-radius:18px;padding:22px;margin:24px 0;'><h2 #F4D7B5;border-radius:18px;padding:22px;margin:24px 0;'><p style='margin:0 0 0px;color:#7b6a58;font-size:14px;line-height:1.6;'><strong style='color:#2F241C;'>Booking Changes:</strong> If you need to
