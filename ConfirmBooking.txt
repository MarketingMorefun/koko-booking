query ConfirmBooking verb=POST {
  api_group = "KOKO Booking"

  input {
    object payload {
      schema {
        int booking_id
      }
    }
  }

  stack {
    function.run create_booking_checkout {
      input = {
        booking_id    : $input.payload.booking_id
        payment_option: "deposit"
        booking_type  : "birthday_party"
      }
    } as $Result
  }

  response = $Result
}
