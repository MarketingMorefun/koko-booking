function expire_stale_holds {
  input {
  }

  stack {
    function.run reconcile_checkout_holds as $Reconciled
    var $cutoff {
      value = "now"
        |to_timestamp
        |add_ms_to_timestamp:-86400000
    }
  
    db.query bookings {
      where = $db.bookings.status == "hold" && $db.bookings.created_at < $cutoff
      return = {type: "list"}
    } as $Candidates
  
    foreach ($Candidates) {
      each as $Candidate {
        db.transaction {
          stack {
            db.get bookings {
              field_name = "id"
              field_value = $Candidate.id
              lock = true
            } as $Booking
          
            conditional {
              if ($Booking.status == "hold" && $Booking.created_at < $cutoff) {
                conditional {
                  if (($Booking.applied_credit_id|first_notnull:0) > 0) {
                    db.get customer_credits {
                      field_name = "id"
                      field_value = $Booking.applied_credit_id
                      lock = true
                    } as $Credit
                  
                    conditional {
                      if ($Credit != null && $Credit.status == "pending" && ($Credit.used_booking_id == $Booking.id || $Credit.used_booking_id == 0)) {
                        db.patch customer_credits {
                          field_name = "id"
                          field_value = $Credit.id
                          data = {status: "available", used_booking_id: 0}
                        } as $ReleasedCredit
                      }
                    }
                  }
                }
              
                db.patch bookings {
                  field_name = "id"
                  field_value = $Booking.id
                  data = {
                    status            : "expired"
                    hold_expires_at_ts: 0
                    payment_url       : ""
                  }
                } as $Expired
              }
            }
          }
        }
      }
    }
  }

  response = {expired_count: $Candidates|count}
}
