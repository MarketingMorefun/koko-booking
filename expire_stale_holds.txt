function expire_stale_holds {
  input {
  }

  stack {
    var $now_ts {
      value = `"now"|to_timestamp:"Australia/Sydney"`
    }

    var $twentyfour_hours_ago {
      value = `$now_ts|add_ms_to_timestamp:-86400000`
    }

    // Only give up on a hold after 24 hours — this must stay aligned with
    // the abandoned-cart reminder emails (sent at 1h and 20h), which rely
    // on the booking still reading status "hold"/"pending_payment" all the
    // way up to 24h.
    //
    // Also sweeps "pending_payment" bookings: a customer only needs to
    // click "Pay" once (not actually finish on Stripe's page) for status
    // to move to pending_payment, so plenty of abandoned carts end up
    // there rather than staying "hold". Without including it here, those
    // records would sit forever and never get cleaned up.
    //
    // hold_expires_at_ts (15 min) is deliberately NOT used here anymore.
    // That field only controls how long this booking blocks the exact
    // same room/slot from other customers during CreateBooking's conflict
    // check — it does not mean "give up on this customer." Using it here
    // was flipping bookings to "expired" within 15 minutes of anyone else
    // starting a new booking, which silently broke the reminder emails.
    db.query bookings {
      where = ($db.bookings.status == "hold" || $db.bookings.status == "pending_payment") && $db.bookings.created_at < $twentyfour_hours_ago
      return = {type: "list"}
    } as $stale_holds

    foreach ($stale_holds) {
      each as $hold {
        db.edit bookings {
          field_name = "id"
          field_value = $hold.id
          data = {status: "expired"}
        } as $updated

        // Release any credit this abandoned booking had reserved
        // (CreateBooking/CreateGroupBooking flip a credit from
        // "available" to "pending" the moment they decide to use it, so
        // a second concurrent hold can't also grab it). Since this
        // booking never paid, put it back to "available" so the customer
        // can actually use it on a future booking instead of it being
        // stranded on a dead hold forever.
        conditional {
          if (($hold.applied_credit_id|first_notnull:0) > 0) {
            db.patch customer_credits {
              field_name = "id"
              field_value = $hold.applied_credit_id
              data = {status: "available"}
            } as $ReleasedCredit
          }
        }
      }
    }
  }

  response = {expired_count: $stale_holds|count}
}
