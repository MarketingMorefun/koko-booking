query BookingSheetExport verb=GET {
  api_group = "KOKO Booking"

  input {
    text key
  }

  stack {
    var $error_message {
      value = null
    }
  
    var $Bookings {
      value = null
    }
  
    conditional {
      if ($input.key == null || $input.key == "" || $input.key != $env.SHEET_EXPORT_KEY) {
        var $error_message {
          value = "Unauthorized"
        }
      }
    
      else {
        db.query bookings {
          return = {type: "list"}
        } as $Bookings
      }
    }
  }

  response = {error_message: $error_message, bookings: $Bookings}
}
